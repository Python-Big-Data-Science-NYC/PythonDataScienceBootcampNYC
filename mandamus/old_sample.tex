	% SLIDE 1: WHAT IS WOM
	\begin{frame}{What is a Writ of Mandamus (WoM)?}
		\begin{block}{Definition}
			A lawsuit that tells the government: \textit{“You have 60 days to respond or come to court.”}
		\end{block}
		\begin{columns}[T]
			\column{0.48\textwidth}
			\textbf{What it IS:}
			\begin{itemize}
				\item For \textbf{stuck} cases (way beyond normal time)
				\item Forces USCIS to make a decision
				\item Speeds things up when nothing else worked
			\end{itemize}
			\column{0.48\textwidth}
			\textbf{What it is NOT:}
			\begin{itemize}
				\item Not an appeal
				\item Not a guaranteed approval
			\end{itemize}
		\end{columns}
		\vspace{0.3cm}
		\textcolor{warningred}{\textbf{When to file:}} No exact rule, but must be OBVIOUSLY taking WAY TOO LONG.\\
		Example: Interview said 1 month → waited 2 years.
	\end{frame}
	
	% SLIDE 2: THE 9 STEPS OVERVIEW
	\begin{frame}{The 9 Steps – Complete Overview}
		\begin{enumerate}
			\setlength\itemsep{0.3cm}
			\item Use AI (Microsoft Copilot) to draft initial WoM – free
			\item Hire Fiverr lawyer to review/edit – approx \$150
			\item Identify \textbf{5 defendants} (USCIS, AG, Director, Mayorkas, your US Attorney)
			\item Attach evidence – congressman replies, service requests, screenshots
			\item Make \textbf{6 identical packets} (complaint + evidence)
			\item File at your federal district court – pay filing fee
			\item Fill out JS 44 (civil cover sheet) + 5 summons forms (AO 440)
			\item Serve defendants by certified mail + email USCIS
			\item File \textbf{proof of service} with court after delivery
		\end{enumerate}
		\vspace{0.2cm}
		\centering
		\textcolor{successgreen}{\textbf{Result:}} \textit{“3 business days later – case approved. Card produced in 5 days.”}
	\end{frame}

		\begin{block}{The 5 Defendants (exactly as on summons)}
			\begin{enumerate}

				\item \textbf{Merrick Garland} – Attorney General of the United States

	% SLIDE 4: SUMMONS FORM (AO 440) – HOW TO FILL
	\begin{frame}{Summons Form (AO 440) – One per Defendant}
		\begin{block}{Step-by-Step for Each Summons}
			\begin{itemize}
				\item \textbf{To:} Write full name and title of the defendant exactly as in your WoM
				\item \textbf{Plaintiff:} Your full name and address
				\item \textbf{Defendant:} Same as above
				\item Leave \textbf{Date} and \textbf{Signature of Clerk} blank – court fills those
			\end{itemize}
		\end{block}
		\textbf{You need 5 summons forms – one for each defendant.}\\
		Attach one stamped summons to each packet after filing.
	\end{frame}
	
	% SLIDE 5: CIVIL COVER SHEET (JS 44) – QUICK GUIDE
	\begin{frame}{Civil Cover Sheet (JS 44) – Only One}
		\begin{columns}[T]
			\column{0.5\textwidth}
			\textbf{Fill these fields:}
			\begin{itemize}
				\item \textbf{Plaintiffs:} Your name (the applicant)
				\item \textbf{County:} Your home county
				\item \textbf{Defendants:} Write “USCIS”
				\item \textbf{County of Defendant:} Prince George’s County (USCIS HQ)
				\item Attorneys part: \textbf{leave blank}
			\end{itemize}
			\column{0.5\textwidth}
			\textbf{Mark X in:}
			\begin{itemize}
				\item Basis of Jurisdiction: \textbf{U.S. Government Defendant}
				\item Nature of Suit: \textbf{465 – Other Immigration Actions}
				\item Origin: \textbf{Original Proceeding}
			\end{itemize}
			Sign \& date at “Signature of Attorney of Record”
		\end{columns}
		\vspace{0.2cm}
		\tiny Citizenship of Principal Parties and Cause of Action – leave blank.
	\end{frame}
	
	% SLIDE 6: EVIDENCE – EXHIBITS A, B, C...
	\begin{frame}{Evidence – Attach to Back of WoM}
		\begin{block}{What to Include (shows you tried everything)}
			\begin{itemize}
				\item Congressman/senator replies
				\item USCIS service requests with generic responses
				\item Emma chat screenshots
				\item CIS Ombudsman request
				\item Original receipt notice (I-797)
			\end{itemize}
		\end{block}
		\textbf{How to organize:}
		\begin{itemize}
			\item Create an “Exhibit A” cover page (big letters in Word)
			\item Behind it, attach that evidence
			\item Repeat for Exhibit B, C, D, etc.
			\item Attach all exhibits to the back of your WoM complaint
		\end{itemize}
		\textcolor{titleblue}{Make 6 identical packets – each has WoM + all exhibits.}
	\end{frame}
	
	% SLIDE 7: SERVING THE DEFENDANTS (STEP 8)
	\begin{frame}{Serving the Defendants – Certified Mail + Email}
		\begin{columns}[T]
			\column{0.5\textwidth}
			\textbf{By Certified Mail / Overnight:}
			\begin{itemize}
				\item One packet per defendant
				\item Attach stamped summons to front
				\item Use addresses from WoM (page 2)
				\item Save ALL tracking numbers
			\end{itemize}
			\column{0.5\textwidth}
			\textbf{Email to USCIS (required):}
			\begin{itemize}
				\item Send to: \texttt{uscis.serviceofprocess@uscis.dhs.gov}
				\item Attach: Summons + WoM + evidence as \textbf{compressed PDF}
				\item Keep auto-reply email as proof
			\end{itemize}
		\end{columns}
		\vspace{0.3cm}
		\textbf{After all delivered:} File \textbf{Proof of Service} with the court.
		\begin{itemize}
			\item Print USPS delivery confirmations
			\item Attach email proof to USCIS
			\item File at clerk’s office
		\end{itemize}
	\end{frame}
	
	% SLIDE 8: PROOF OF SERVICE OUTLINE
	\begin{frame}{Proof of Service – What to File}
		\begin{block}{Document Structure}
			\begin{enumerate}
				\item Caption: Same as your WoM (case number already assigned)
				\item Title: “PROOF OF SERVICE”
				\item For each defendant, state:
				\begin{itemize}
					\item Name and title
					\item Method of service (certified mail / overnight)
					\item Tracking number
					\item Date delivered
					\item Attach USPS delivery confirmation printout
				\end{itemize}
				\item For USCIS: also state email sent + attach auto-reply
				\item Sign under penalty of perjury
			\end{enumerate}
		\end{block}
		\tiny After filing Proof of Service, the court will track the 60-day response deadline.
	\end{frame}

	% SLIDE 10: AFTER APPROVAL – MOTION TO DISMISS
	\begin{frame}{After USCIS Approves – Motion to Dismiss}
		\begin{block}{Once you have your green card / approval in hand}
			\begin{itemize}
				\item You must file a \textbf{Motion to Dismiss} with the court
				\item You do NOT need to mail/serve defendants again – court does that
				\item Use the outline from OP’s Google Drive link
				\item Wait until card \textit{physically in hand} before dismissing
			\end{itemize}
		\end{block}
		\textbf{If USCIS asks for an extension (before approval):}
		\begin{itemize}
			\item You can consent or refuse
			\item If you refuse and they miss 60-day deadline → file Motion for Default
		\end{itemize}
	\end{frame}
