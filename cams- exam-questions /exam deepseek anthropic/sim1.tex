	\begin{frame}{\normalsize DORA (Q5)}
		
		\hardexplain{Students confuse DORA with GDPR. KEY = threat-led, not compliance-driven.}
		
		\hard{Requires THREAT-LED PENETRATION TESTING (TLPT) -- NOT standard vuln scans.}

		\textbf{Why Advanced Testing:}
		\begin{table}
			\begin{tabular}{|p{3.8cm}|p{3.8cm}|}
				\hline
				\textbf{Traditional} & \textbf{DORA Advanced} \\
				\hline
				Checks compliance with known standards & Simulates real cyber attacks \\
				\hline
				Passive checking & Active threat-led testing \\
				\hline
				"Do we have antivirus?" & "Will the system survive?" \\
				\hline
			\end{tabular}
		\end{table}
		
		\textbf{Tests:} Basic (vuln scans) $\to$ Advanced (pen testing) $\to$ \textcolor{red}{TLPT (mandatory 3yrs)}
		
		\scenario{Bank passes compliance but fails ransomware.}
		
		\trap{Not "Because regulators require it." Purpose = detect vulnerabilities BEFORE incidents.}
		
		\correct{To detect potential vulnerabilities and assess system defences before incidents occur.}
	\end{frame}
