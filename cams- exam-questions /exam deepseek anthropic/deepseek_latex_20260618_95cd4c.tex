	% ----- SLIDE 16 -----
	\begin{frame}{\normalsize Rec. 35 -- EPD}
		
		\textbf{EPD = Effective, Proportionate, Dissuasive}
		\begin{enumerate}
			\item \textbf{Effective}: actually deters
			\item \textbf{Proportionate}: fits severity
			\item \textbf{Dissuasive}: discourages future
		\end{enumerate}
		
		\textbf{Sanctions:} Criminal (imprisonment, fines), Civil (penalties, damages), Administrative (license, cease, debarment)
		
		\textbf{ALL THREE against BOTH individuals AND legal persons.}
		
		\scenario{\$500 fine for \$50M AML failure.}
		
		\trap{Fine exists but is it DISSUASIVE? NO! \$500 for \$50M violates EPD.}
		
		\correct{Criminal, civil, AND administrative sanctions against both individuals and legal persons.}
	\end{frame}

			\end{tabular}
		\end{table}
		
		\textbf{Independence:}
		\begin{itemize}
			\item Third line CANNOT design controls it tests

		\end{itemize}

		\trap{This violates independence -- not efficient.}
		
		\correct{Audit cannot design controls it later tests.}
	\end{frame}

	\begin{frame}{\normalsize OFAC 50\% Rule}
		
		\textbf{Principle:}
		\begin{itemize}
			\item Ownership calculated by multiplying through tiers
			\item ONLY blocked entities transmit blocked status downstream
		\end{itemize}
		
		\textbf{Example:}
		\begin{itemize}
			\item X (SDN) owns 55\% of A
			\item A owns 40\% of B
			\item B owns 60\% of C
		\end{itemize}
		
		\begin{table}
			\begin{tabular}{|p{2.5cm}|p{5cm}|}
				\hline
				\textbf{Entity} & \textbf{Status} \\
				\hline
				A & \textcolor{red}{BLOCKED} (55\% > 50\%) \\
				\hline
				B & NOT BLOCKED (40\% < 50\%) \\
				\hline
				C & NOT BLOCKED (B not blocked, so 60\% does NOT transmit) \\
				\hline
			\end{tabular}
		\end{table}
		
		\trap{C NOT blocked even though B owns 60\%. Since B not blocked, cannot transmit.}
		
		\correct{C is NOT blocked. Only blocked entities transmit blocked status.}
	\end{frame}
