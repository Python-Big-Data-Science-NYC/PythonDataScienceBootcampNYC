	% ============ Q201 ============
	\begin{frame}[t]{Q201}
		\small
		\textbf{Question:} An auditor is assigned to audit the same department for the third consecutive year. Which is the MOST likely concern?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item The auditor may lose objectivity through familiarity.
			\item The auditor will be too efficient.
			\item The auditor will find too many issues.
			\item The auditor will need less documentation.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: A}
	\end{frame}
	
	\begin{frame}[t]{Q201 -- Explanation}
		\small
		\textbf{Correct: A.} Repeated assignments create a familiarity threat; rotation or safeguards may be needed. \\[4pt]
		\textbf{B wrong:} Efficiency is not a concern. \\[2pt]
		\textbf{C wrong:} Findings are not a concern. \\[2pt]
		\textbf{D wrong:} Documentation is always required.
	\end{frame}
