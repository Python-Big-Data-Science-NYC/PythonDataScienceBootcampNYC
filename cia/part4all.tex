	% ============ Q301 ============
	\begin{frame}[t]{Q301}
		\small
		\textbf{Question:} An auditor is reviewing a cloud service provider's SOC 2 Type II report. Which is the MOST important consideration?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item The report covers a period, so the auditor must assess whether the period is recent enough and whether complementary user entity controls (CUECs) are implemented.
			\item The report guarantees the provider is secure.
			\item The report replaces the need for any internal audit testing.
			\item The report is only relevant to external audit.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: A}
	\end{frame}
	
	\begin{frame}[t]{Q301 -- Explanation}
		\small
		\textbf{Correct: A.} A SOC 2 Type II covers a period; the auditor must evaluate the period's relevance and confirm the client implemented the CUECs. \\[4pt]
		\textbf{B wrong:} No report guarantees security. \\[2pt]
		\textbf{C wrong:} SOC reports inform, but do not eliminate, internal audit testing. \\[2pt]
		\textbf{D wrong:} SOC reports are relevant to internal audit too.
	\end{frame}
