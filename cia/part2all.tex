	% ============ Q101 ============
	\begin{frame}[t]{Q101}
		\small
		\textbf{Question:} The CAE is asked to lead the implementation of a new enterprise risk management (ERM) framework. Which response BEST preserves conformance with the Standards?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Accept, because ERM is a core internal audit competency.
			\item Decline all ERM involvement to protect independence.
			\item Accept only an advisory role, ensuring management retains ownership and decisions.
			\item Accept and later audit the framework as assurance.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: C}
	\end{frame}
	
	\begin{frame}[t]{Q101 -- Explanation}
		\small
		\textbf{Correct: C.} Internal audit may consult on ERM but must not own it or make management decisions; doing so would impair objectivity and preclude later assurance. \\[4pt]
		\textbf{A wrong:} Competency is not the issue; ownership is. Taking the lead makes internal audit management. \\[2pt]
		\textbf{B wrong:} Total abstention is overly restrictive; advisory support is permitted. \\[2pt]
		\textbf{D wrong:} Implementing then auditing the same framework is a self-review threat (Standard 1130).
	\end{frame}

		\textbf{Question:} An auditor is assigned to audit the payroll function. The auditor's spouse is the payroll supervisor. Which action is MOST appropriate?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Proceed, since the spouse does not approve payments.
			\item Disclose the relationship and request reassignment.
			\item Proceed but avoid interviewing the spouse.
			\item Proceed and document the relationship in the workpapers only.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: B}
	\end{frame}
	
	\begin{frame}[t]{Q102 -- Explanation}
		\small
		\textbf{Correct: B.} A close family relationship with the auditee supervisor is an objectivity impairment; the auditor must disclose and be reassigned (Standard 1130). \\[4pt]
		\textbf{A wrong:} The impairment exists regardless of approval authority. \\[2pt]
		\textbf{C wrong:} Avoiding interviews does not remove the impairment. \\[2pt]
		\textbf{D wrong:} Documentation alone does not cure an impairment; reassignment is required.
	\end{frame}

		\textbf{Question:} The auditor most likely increases the sample size when:
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Tolerable deviation rate increases.
			\item Expected deviation rate increases.
			\item Population size shrinks slightly.
			\item Confidence level decreases.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: B}
	\end{frame}
	
	\begin{frame}[t]{Q111 -- Explanation}
		\small

		\textbf{A wrong:} Higher tolerable rate \emph{reduces} required sample size. \\[2pt]
		\textbf{C wrong:} Small population changes rarely matter much. \\[2pt]
		\textbf{D wrong:} Lower confidence reduces sample size.
	\end{frame}
