	% ============ Q401 ============
	\begin{frame}[t]{Q401}
		\small
		\textbf{Question:} An internal auditor is asked by a friend in the auditee department to "give a heads-up" about which samples will be selected. Which action BEST conforms to the IIA Code of Ethics?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Share the sample list, since it is not confidential.
			\item Decline, because it compromises the audit's integrity and could allow manipulation of evidence.
			\item Share only the sample size, not the specific items.
			\item Share the list but document the request.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: B}
	\end{frame}
	
	\begin{frame}[t]{Q401 -- Explanation}
		\small
		\textbf{Correct: B.} Pre-disclosing samples lets the auditee prepare or alter evidence, undermining audit integrity (Integrity and Objectivity principles). \\[4pt]
		\textbf{A wrong:} Sample selections are confidential engagement information. \\[2pt]
		\textbf{C wrong:} Even the size can be used to game the process. \\[2pt]
		\textbf{D wrong:} Documenting the breach does not cure it.
	\end{frame}
	
	% ============ Q402 ============
	\begin{frame}[t]{Q402}
		\small
		\textbf{Question:} Which is the BEST example of applying the Competency principle of the IIA Code?
		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Accepting an engagement in a highly technical area without training.
			\item Declining an engagement and referring it to someone with the required expertise.
			\item Relying on the auditee's experts to perform the audit.
			\item Skipping continuing professional education to save time.
		\end{enumerate}
		\vspace{4pt}
		\textbf{Correct Answer: B}
	\end{frame}
	
	\begin{frame}[t]{Q402 -- Explanation}
		\small
		\textbf{Correct: B.} Competency requires the knowledge, skills, and experience needed; if absent, decline or obtain competent help (Standard 1210, Code Rule 4.1). \\[4pt]
		\textbf{A wrong:} Accepting without competence violates the Code. \\[2pt]
		\textbf{C wrong:} The auditee cannot substitute for auditor competence. \\[2pt]
		\textbf{D wrong:} Skipping CPE undermines proficiency (Standard 1230).
	\end{frame}

		\begin{enumerate}[label=\Alph*.,itemsep=1pt,topsep=2pt]
			\item Accept; speaking fees are not gifts.
			\item Decline or disclose and evaluate whether it impairs objectivity, since it creates a self-interest threat.

		\textbf{D wrong:} "Small" is subjective and still creates a threat.
