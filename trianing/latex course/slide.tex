	\begin{frame}{Workflow Overview}
		\begin{enumerate}
			\item \textbf{Identify Latest Version}
			\begin{itemize}
				\item Check Git repository: \texttt{develop} vs \texttt{master} branch
				\item Determine which is most recent
			\end{itemize}
			\item \textbf{Convert latex from doc  Working Version}
			\begin{itemize}
				\item Table errors detected
				\item Roll back to older stable code
			\end{itemize}
			\item \textbf{Convert Word Document}
			\begin{itemize}
				\item Transform \texttt{.docx} to \LaTeX{} format
			\end{itemize}
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Attachment \& Visual Management}
		\begin{columns}
			\column{0.5\textwidth}
			\textbf{Attachments}
			\begin{itemize}
				\item Enterprise \LaTeX{} Git code lacks attachment support
				\item Move attachments to SharePoint folders
				\item Update references accordingly
			\end{itemize}
			\column{0.5\textwidth}
			\textbf{Visual Matching}
			\begin{itemize}
				\item Visual consistency is critical
				\item Match output appearance precisely
				\item "Eyes are important"
			\end{itemize}
		\end{columns}
	\end{frame}
