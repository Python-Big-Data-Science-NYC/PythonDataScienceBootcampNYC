	\begin{frame}{Manual Indexing: Step-by-Step Process}
		\begin{enumerate}
			\item \textbf{Pre-Submission Preparation} (We Do This)
			\begin{itemize}
				\item Format paper to each platform's requirements
				\item Prepare metadata (abstract, keywords, JEL codes)
				\item Create ORCID \& institutional affiliations
			\end{itemize}
			
			\item \textbf{Parallel Submissions} (We Do This)
			\begin{itemize}
				\item Submit to SSRN \& TechArxiv \textbf{same day}
				\item Register on OSF \& create project page
				\item Submit to HAL \& MPRA within 48 hours
			\end{itemize}
			
			\item \textbf{Post-Submission Tracking} (We Do This)
			\begin{itemize}
				\item Monitor approval/rejection status
				\item Update metadata if required
				\item Ensure persistent URLs (DOIs)
			\end{itemize}
			
			\item \textbf{You Do This}
			\begin{itemize}
				\item Review final submissions
				\item Approve author proofs
				\item Share paper links on your social media
			\end{itemize}
		\end{enumerate}
