	\section{Question 1: KRIs Maintenance}
	
	\begin{frame}{Q1: Why Maintain KRIs?}
		\begin{block}{Question}
			Which of the following is the MOST important reason to maintain key risk indicators (KRIs)?
		\end{block}
		\begin{enumerate}[(A)]
			\item In order to avoid risk
			\item Complex metrics require fine-tuning
			\item Risk reports need to be timely
			\item Threats and vulnerabilities change over time
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Q1: Explanation}
		\begin{block}{Correct Answer: D}
			Threats and vulnerabilities change over time and KRI maintenance ensures that KRIs continue to effectively capture these changes.
		\end{block}
		\begin{block}{Option Analysis}
			\begin{itemize}
				\item \textbf{A - Incorrect:} Risk avoidance is a risk response, not a reason for KRI maintenance
				\item \textbf{B - Incorrect:} Fine-tuning is secondary; capturing changes is primary
				\item \textbf{C - Incorrect:} Timeliness is a business requirement, not a maintenance reason
				\item \textbf{D - Correct:} Risk environment is dynamic; KRIs must evolve with threats
			\end{itemize}
		\end{block}
	\end{frame}

	\section{Question 5: Risk Priority Number}
	
	\begin{frame}{Q5: RPN Calculation}
		\begin{block}{Question}
			A risk has rating for occurrence, severity, and detection as 4, 5, and 6 respectively. What Risk Priority Number (RPN) would you give?
		\end{block}
		\begin{enumerate}[(A)]
			\item 120
			\item 100
			\item 15
			\item 30
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Q5: Explanation}
		\begin{block}{Correct Answer: A}
			RPN = Severity × Occurrence × Detection = 4 × 5 × 6 = 120
		\end{block}
		\begin{block}{Option Analysis}
			\begin{itemize}
				\item \textbf{A - Correct:} 4 × 5 × 6 = 120
				\item \textbf{B - Incorrect:} Would be sum of values multiplied incorrectly
				\item \textbf{C - Incorrect:} This would be addition (4+5+6=15)
				\item \textbf{D - Incorrect:} This ignores one factor (4×5=20 or 5×6=30)
			\end{itemize}
		\end{block}
	\end{frame}

	\section{Question 55: Exposure Factor Value} \begin{frame}{Q55: Complete Asset Loss} \begin{block}{Question} What is the value of exposure factor if the asset is lost completely? \end{block} \begin{enumerate}[(A)] \item 1 \item Infinity \item 10 \item 0 \end{enumerate} \end{frame} \begin{frame}{Q55: Key Concept} \begin{block}{Exposure Factor (EF)} Exposure Factor represents the percentage of an asset's value lost as a result of a single risk event. \end{block} \[ EF = \frac{\text{Loss Amount}}{\text{Asset Value}} \] \begin{itemize} \item EF ranges from 0 to 1 \item 0 = No loss (0\%) \item 0.5 = Partial loss (50\%) \item 1 = Complete loss (100\%) \end{itemize} \end{frame} \begin{frame}{Q55: Relationship to SLE} \begin{block}{Single Loss Expectancy (SLE)} Exposure Factor is used to calculate Single Loss Expectancy. \end{block} \[ SLE = AV \times EF \] where: \begin{itemize} \item AV = Asset Value \item EF = Exposure Factor \item SLE = Expected loss from one occurrence \end{itemize} \end{frame} \begin{frame}{Q55: Example} \begin{block}{Example} A server worth \$100,000 is completely destroyed by fire. \end{block} \[ AV = \$100{,}000 \] \[ EF = 1.0 \] \[ SLE = \$100{,}000 \times 1.0 \] \[ SLE = \$100{,}000 \] Since the entire asset is lost, the Exposure Factor is 1 (100\%). \end{frame} \begin{frame}{Q55: Option Analysis} \begin{block}{Correct Answer: A} Exposure Factor equals 1.0 when an asset suffers a complete loss (100\% impact). \end{block} \begin{itemize} \item \textbf{A - Correct:} EF = 1 means 100\% of asset value is lost. \item \textbf{B - Incorrect:} EF cannot be infinite; it is expressed as a percentage from 0 to 1. \item \textbf{C - Incorrect:} 10 would imply 1000\% loss, which is impossible for a single asset. \item \textbf{D - Incorrect:} EF = 0 means no loss occurred. \end{itemize} \end{frame} \begin{frame}{Q55: Exam Tip} \begin{alertblock}{Remember} If the asset is completely destroyed, stolen, or rendered unusable, Exposure Factor = 1.0 (100\%). If half the value is lost, Exposure Factor = 0.5 (50\%). \end{alertblock} \end{frame}

\section{Final Review} \begin{frame}{CRISC Key Takeaways} \begin{block}{Risk Management Fundamentals} \begin{itemize} \item Risk management is an ongoing and iterative process. \item Risks must be identified, analyzed, assessed, monitored, and communicated continuously. \item New threats, vulnerabilities, technologies, and business objectives can create new risks at any time. \item Effective risk management aligns business objectives with risk appetite and risk tolerance. \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Key Takeaways} \begin{block}{Controls and Objective Achievement} \begin{itemize} \item Controls exist to support achievement of business objectives. \item A control should be evaluated based on effectiveness, efficiency, and risk reduction. \item Strong controls with poor business alignment may still fail to support organizational goals. \item Control assessments should determine whether residual risk remains within acceptable limits. \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Key Takeaways} \begin{block}{Key Risk Indicators (KRIs)} \begin{itemize} \item KRIs provide early warning signs of increasing risk. \item KRIs help management identify unfavorable trends before major incidents occur. \item KRIs are predictive and forward looking. \item Examples: \begin{itemize} \item Increase in security incidents \item Growing number of policy violations \item Rising employee turnover \item Increase in system downtime \end{itemize} \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Key Takeaways} \begin{block}{Negative Risk Responses (Threats)} \begin{itemize} \item Avoid: Eliminate the risk completely. \item Mitigate: Reduce likelihood or impact. \item Transfer: Shift risk to another party (insurance, outsourcing). \item Accept: Take no additional action and knowingly retain the risk. \item Risk ownership remains with the business. \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Key Takeaways} \begin{block}{Positive Risk Responses (Opportunities)} \begin{itemize} \item Exploit: Ensure the opportunity occurs. \item Enhance: Increase probability or benefit. \item Share: Partner with another party to maximize benefit. \item Accept: Take advantage of the opportunity if it occurs. \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Key Takeaways} \begin{block}{Quantitative Risk Analysis} \[ SLE = AV \times EF \] \[ ALE = SLE \times ARO \] Where: \begin{itemize} \item AV = Asset Value \item EF = Exposure Factor \item SLE = Single Loss Expectancy \item ARO = Annualized Rate of Occurrence \item ALE = Annualized Loss Expectancy \end{itemize} \textbf{Example:} Asset Value = \$100,000 Exposure Factor = 40\% ARO = 2 SLE = \$40,000 ALE = \$80,000 \end{block} \end{frame} \begin{frame}{CRISC Exam Essentials} \begin{block}{Most Important Exam Principle} \centering \Large \textbf{Business Management Owns Risk} \end{block} \begin{itemize} \item Business management owns and accepts risk. \item The CRO provides oversight. \item IT supports business objectives. \item Security teams implement controls. \item Risk decisions belong to management. \end{itemize} \end{frame} \begin{frame}{CRISC Exam Essentials} \begin{block}{Data Owner vs Data Custodian} \textbf{Data Owner} \begin{itemize} \item Classifies information \item Approves access \item Reviews user entitlements \item Determines protection requirements \end{itemize} \textbf{Data Custodian} \begin{itemize} \item Implements controls \item Maintains systems \item Performs backups \item Administers security tools \end{itemize} \end{block} \end{frame} \begin{frame}{CRISC Exam Essentials} \begin{block}{CIA Triad} \begin{itemize} \item \textbf{Confidentiality} Protect information from unauthorized disclosure. \item \textbf{Integrity} Protect information from unauthorized modification. \item \textbf{Availability} Ensure information is available when needed. \end{itemize} Most security controls support one or more elements of the CIA Triad. \end{block} \end{frame} \begin{frame}{CRISC Exam Essentials} \begin{block}{Capability Maturity Levels} \begin{itemize} \item Level 0 – Nonexistent \item Level 1 – Initial / Ad Hoc \item Level 2 – Repeatable \item Level 3 – Defined \item Level 4 – Managed and Measurable \item Level 5 – Optimized \end{itemize} \textbf{Exam Tip:} If you see "continuous improvement," "optimization," or "best practices," think Level 5. \end{block} \end{frame} \begin{frame}{CRISC Exam Frameworks}
