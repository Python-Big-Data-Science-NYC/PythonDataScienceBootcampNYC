	\begin{frame}{Q201: Multi-Risk Response Interaction}
		\begin{block}{Question}
			A risk response for Risk A creates a new secondary risk. The secondary risk is assessed as high. What should happen?
		\end{block}
		\begin{enumerate}[(A)]
			\item Ignore secondary risk - focus on primary risk
			\item Assess and treat secondary risk as a new risk
			\item Revert the original risk response
			\item Transfer the secondary risk
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Q201: Explanation}
		\begin{block}{Correct Answer: B}
			\textbf{Secondary risks must be assessed and treated:}
			\begin{itemize}
				\item Secondary risk = new risk created by risk response
				\item Must be entered into risk register
				\item Assessed for likelihood and impact
				\item Treated like any other risk
			\end{itemize}
		\end{block}
		\begin{alertblock}{Critical Trap}
			Risk response can create new risks.
			Always assess secondary risks.
		\end{alertblock}
	\end{frame}
	
	\begin{frame}{Q202: Risk Response Cost-Benefit Edge Case}
		\begin{block}{Question}
			A control costs \$100K and reduces risk from \$150K to \$60K. However, the risk has only 10\% likelihood. Should the control be implemented?
		\end{block}
		\begin{enumerate}[(A)]
			\item Yes, because residual risk is lower
			\item No, because expected loss reduction is less than control cost
			\item Yes, because risk exceeds appetite
			\item Cannot determine
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Q202: Explanation}
		\begin{block}{Correct Answer: B}
			\textbf{Calculation:}
			\begin{itemize}
				\item Expected loss before = 10\% × \$150K = \$15K
				\item Expected loss after = 10\% × \$60K = \$6K
				\item Risk reduction = \$9K
				\item Control cost = \$100K
				\item \$100K > \$9K = NOT justified
			\end{itemize}
		\end{block}
		\begin{alertblock}{Critical Trap}
			Use expected loss (likelihood × impact), not just impact.
		\end{alertblock}
	\end{frame}

	\begin{frame}{Q206: Mitigation Implementation Delay}
		\begin{block}{Question}
			A control takes 6 months to implement. The risk has 80\% likelihood and \$500K impact. What is the risk during implementation?
		\end{block}
		\begin{enumerate}[(A)]
			\item No risk
			\item \$200K expected loss
			\item \$400K expected loss
			\item \$500K expected loss
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Q206: Explanation}
		\begin{block}{Correct Answer: B}
			\textbf{Calculation:}
			\begin{itemize}
				\item Expected annual loss = 80\% × \$500K = \$400K
				\item During 6 months = \$400K × 0.5 = \$200K
				\item Risk continues during implementation
			\end{itemize}
		\end{block}
		\begin{alertblock}{Critical Trap}
			Implementation period = Continued exposure.
			Consider interim controls.
		\end{alertblock}
	\end{frame}
